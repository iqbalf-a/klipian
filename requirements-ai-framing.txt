# klipian -- ADDITIONAL dependencies for AI Framing (automatic speaker detection)
#
# Optional: only needed if you use the "AI Framing" button on the Framing
# screen. Pulls in PyTorch (~hundreds of MB) -- that's why it's split out
# from the main requirements.txt instead of just being merged in.
#
# Needs HF_TOKEN once at the start to download the model weights -- see
# .env.example and the comment in klipian/diarize.py.
#
#   pip install -r requirements.txt -r requirements-ai-framing.txt

pyannote.audio>=4.0.0      # speaker diarization (CPU, via torch/torchaudio)
soundfile>=0.12.0          # manual WAV read -- workaround for a torchcodec bug on Windows
# Face detection (YuNet, an ONNX model vendored at assets/models/ -- NOT
# PyTorch) for tightening the crop box in AI Framing/manual framing points.
# Version kept < 5.0 as a conservative choice, not a hard requirement like
# it used to be: face detection used to run on Haar cascades, and 5.x
# dropped the bundled cascade XML files entirely (an upstream packaging
# change) -- that specific breakage no longer applies now that detection
# is YuNet (a vendored ONNX file, not something cv2 needs to bundle). The
# pin stays for now simply because 5.x hasn't been tested against this
# codebase yet, not because of a known incompatibility.
opencv-python-headless>=4.9.0,<5.0.0
